void demo_of_printf_scanf(){
	char rxByte;
	printf("Give Red LED control input (Y = On, N = off):\r\n");
	scanf ("%c", &rxByte);
	if (rxByte == 'N' || rxByte == 'n'){
		LED_Off();
		printf("LED is Off\r\n\r\n");
	}
	else if (rxByte == 'Y' || rxByte == 'y'){
		LED_On();
		printf("LED is On\r\n\r\n");
	}
}

void demo_of_UART_print(int a){
	int n, i;
	float b;
	b = (float)a/100;
	n = sprintf((char *)buffer, "a = %d\t", a);
	n += sprintf((char *)buffer + n, "b = %f\r\n", b);
	USART_Write(USART2, buffer, n);		
	for (i = 0; i < 8000000; i++); // Delay
	LED_Toggle();
}
	
int main(void){

	char str[64];

	System_Clock_Init(); // Switch System Clock = 80 MHz
	LED_Init();
	UART2_Init();
	Keypad_Pin_Init();

	while (1){
		// demo_of_printf_scanf();
		// demo_of_UART_print(a);
		// a++;

		chr = keypad_scan();
		
		if (chr == '*'){
			if( n >= 1 ) n--;
			str[n] = 0x00; // Remove one character
			USART_Write(USART2, (uint8_t *)str, n);	
			LED_Toggle();
		} else if (chr != 0xFF) {
			str[n] = chr;
			str[n+1] = 0x00; //End of string
			n++;
			if( n >= 62 ) n = 0; //Avoid overflow
			USART_Write(USART2, (uint8_t *)str, n);
			LED_Toggle();
		}
	}
}
